cmake_minimum_required(VERSION 3.16)
project(QLBox VERSION 0.6 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(QLBOX_SDL "Build the SDL2 window frontend" ON)
add_library(qlcore STATIC src/debugger.c src/mouse.c src/qsound.c src/beeper.c src/terminal.c src/hostfs.c src/imagefs.c src/ql.c vendor/musashi/m68kcpu.c
    vendor/musashi/m68kops.c vendor/musashi/softfloat/softfloat.c)
target_include_directories(qlcore PUBLIC src vendor/musashi)
if(NOT MSVC)
    target_link_libraries(qlcore PUBLIC m)
endif()
if(WIN32)
 target_link_libraries(qlcore PUBLIC ws2_32)
endif()
add_executable(qlrun src/main.c)
target_link_libraries(qlrun PRIVATE qlcore)
add_executable(qlbox-headless src/main.c)
target_link_libraries(qlbox-headless PRIVATE qlcore)
if(QLBOX_SDL)
    find_package(SDL2 REQUIRED)
    add_executable(qlbox src/main.c src/sdl_input.c)
    target_compile_definitions(qlbox PRIVATE QLBOX_SDL SDL_MAIN_HANDLED)
    if(TARGET SDL2::SDL2)
        target_link_libraries(qlbox PRIVATE qlcore SDL2::SDL2)
    else()
        target_include_directories(qlbox PRIVATE ${SDL2_INCLUDE_DIRS})
        target_link_libraries(qlbox PRIVATE qlcore ${SDL2_LIBRARIES})
    endif()
    add_executable(qlbox-input-test tests/sdl_input_test.c src/sdl_input.c)
    target_link_libraries(qlbox-input-test PRIVATE qlcore)
    target_compile_definitions(qlbox-input-test PRIVATE SDL_MAIN_HANDLED)
    if(MSVC)
        target_compile_options(qlbox-input-test PRIVATE /UNDEBUG)
    else()
        target_compile_options(qlbox-input-test PRIVATE -UNDEBUG)
    endif()
    if(TARGET SDL2::SDL2)
        target_link_libraries(qlbox-input-test PRIVATE SDL2::SDL2)
    else()
        target_include_directories(qlbox-input-test PRIVATE ${SDL2_INCLUDE_DIRS})
        target_link_libraries(qlbox-input-test PRIVATE ${SDL2_LIBRARIES})
    endif()
    add_executable(qlbox-mouse-test tests/mouse_test.c src/sdl_input.c)
    target_link_libraries(qlbox-mouse-test PRIVATE qlcore)
    target_compile_definitions(qlbox-mouse-test PRIVATE SDL_MAIN_HANDLED)
    if(MSVC)
        target_compile_options(qlbox-mouse-test PRIVATE /UNDEBUG)
    else()
        target_compile_options(qlbox-mouse-test PRIVATE -UNDEBUG)
    endif()
    if(TARGET SDL2::SDL2)
        target_link_libraries(qlbox-mouse-test PRIVATE SDL2::SDL2)
    else()
        target_include_directories(qlbox-mouse-test PRIVATE ${SDL2_INCLUDE_DIRS})
        target_link_libraries(qlbox-mouse-test PRIVATE ${SDL2_LIBRARIES})
    endif()
    # Requires a real/virtual display (SDL dummy does not support capture).
    enable_testing()
    add_test(NAME sdl_input COMMAND qlbox-input-test)
    set_tests_properties(sdl_input PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
endif()
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/rom")
configure_file(rom/js-rom.bin rom/js-rom.bin COPYONLY)

    set_tests_properties(rom_boot_and_basic PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
endif()
